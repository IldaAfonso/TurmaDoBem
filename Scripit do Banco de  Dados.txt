DROP TABLE atendimentos CASCADE CONSTRAINTS;
DROP TABLE beneficiarios CASCADE CONSTRAINTS;
DROP TABLE profissionais CASCADE CONSTRAINTS;
DROP TABLE campanhas CASCADE CONSTRAINTS;
DROP TABLE patrocinadores CASCADE CONSTRAINTS;

CREATE TABLE profissionais (
    id NUMBER PRIMARY KEY,
    nome VARCHAR2(100) NOT NULL,
    telefone VARCHAR2(20),
    cidade VARCHAR2(100),
    tipo VARCHAR2(30)
);

CREATE TABLE beneficiarios (
    id NUMBER PRIMARY KEY,
    nome VARCHAR2(100) NOT NULL,
    telefone VARCHAR2(20),
    cidade VARCHAR2(100),
    necessidade VARCHAR2(200),
    prioridade VARCHAR2(20),
    profissional_indicado VARCHAR2(30)
);


CREATE TABLE atendimentos (
    id NUMBER PRIMARY KEY,
    beneficiario_id NUMBER,
    profissional_id NUMBER,
    data_atendimento DATE,
    observacoes VARCHAR2(300),

    CONSTRAINT fk_beneficiario
    FOREIGN KEY (beneficiario_id)
    REFERENCES beneficiarios(id),

    CONSTRAINT fk_profissional
    FOREIGN KEY (profissional_id)
    REFERENCES profissionais(id)
);


CREATE TABLE campanhas (
    id NUMBER PRIMARY KEY,
    nome_campanha VARCHAR2(100),
    cidade VARCHAR2(100),
    status VARCHAR2(30)
);


CREATE TABLE patrocinadores (
    id NUMBER PRIMARY KEY,
    nome_empresa VARCHAR2(100),
    cidade VARCHAR2(100),
    valor_doacao NUMBER(10,2)
);

--------------------------------------------------
-- Relatório com classificação
--------------------------------------------------

SELECT *
FROM profissionais
ORDER BY nome ASC;

--------------------------------------------------
-- Relatório com função numérica
--------------------------------------------------

SELECT nome_empresa,
       ROUND(valor_doacao,0) AS valor_arredondado
FROM patrocinadores;

--------------------------------------------------
-- Relatório com função de grupo
--------------------------------------------------

SELECT tipo,
       COUNT(*) AS quantidade
FROM profissionais
GROUP BY tipo;

--------------------------------------------------
-- Relatório com subconsulta
--------------------------------------------------

SELECT *
FROM profissionais
WHERE cidade =
(
    SELECT cidade
    FROM beneficiarios
    GROUP BY cidade
    HAVING COUNT(*) =
    (
        SELECT MAX(COUNT(*))
        FROM beneficiarios
        GROUP BY cidade
    )
);


--------------------------------------------------
-- Relatório com junção de tabelas
--------------------------------------------------

SELECT
    b.nome AS beneficiario,
    p.nome AS profissional,
    p.tipo,
    a.data_atendimento
FROM atendimentos a
INNER JOIN beneficiarios b
ON a.beneficiario_id = b.id
INNER JOIN profissionais p
ON a.profissional_id = p.id;

--------------------------------------------------
-- Beneficiários com prioridade urgente

SELECT nome,
       cidade,
       necessidade
FROM beneficiarios
WHERE prioridade = 'Urgente';

--------------------------------------------------
-- Relatório utilizando subconsulta
-- Beneficiários da cidade com maior demanda
--------------------------------------------------

SELECT
    nome,
    cidade,
    necessidade
FROM beneficiarios
WHERE cidade =
(
    SELECT cidade
    FROM beneficiarios
    GROUP BY cidade
    HAVING COUNT(*) =
    (
        SELECT MAX(COUNT(*))
        FROM beneficiarios
        GROUP BY cidade
    )
);
